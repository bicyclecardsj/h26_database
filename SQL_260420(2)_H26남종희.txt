(1)학생 수강 전체 현황 뷰를 작성하시오.
(학생정보+수강 과목+성적을 한 분에 확인가능하도록 작성)

create view v_current_enroll as
select s.학번, s.이름, s.전공, s.학년, c.과목이름 as 수강과목, e.성적
from 학생 s
left join 수강 e on s.학번 = e.학번
left join 과목 c on c.과목코드 = e.과목코드;

(2)각 학생별 평균 성적 뷰를 작성하시오.
(각 학생의 전체 수강 과목 평균 성적 계산 뷰)

create view v_student_avg_score as
select s.이름, avg(e.성적) as 평균성적
from 학생 s
join 수강 e on e.학번 = s.학번
group by s.이름

(3)과목별 수강 인원 및 평균 성적 뷰를 작성하시오.
(각 과목의 수강 인원과 평균, 최고,최저 성적 통계 뷰)

create view v_count_enroll_avg_score as
select c.과목이름, count(e.학번) as "수강 인원", avg(e.성적) as 평균, max(e.성적) as 최고, min(e.성적) as 최저
from 과목 c
join 수강 e on c.과목코드 = e.과목코드
group by c.과목이름;

(4)성적 우수 학생 뷰를 작성하시오.(평균 90점 이상)
(평균 성적이 90점 이상이 우수 학생만 조회하는 뷰)

create view v_a_grade_student as
select s.학번, s.이름, avg(e.성적) as "평균 성적"
from 학생 s
join 수강 e on e.학번 = s.학번
group by s.학번, s.이름
having avg(e.성적) >= 90;

(5)전공 별 수강 통계 뷰를 작성하시오.
(전공별 총 수강 건수와 전공 평균 성적 비교 뷰)

create view v_classenroll_avgscore as
select s.전공, count(e.학번) as "총 수강 건수", avg(e.성적) as "전공 평균 성적"
from 학생 s
join 수강 e on s.학번 = e.학번
group by s.전공;

(6)학기별 수강 현황 뷰를 작성하시오.

create view v_sems_current_enroll as
select e.수강학기, e.과목코드, count(e.학번) as 수강인원, c.과목이름
from 수강 e
join 과목 c on c.과목코드 = e.과목코드
group by e.수강학기, e.과목코드, c.과목이름;

(7)미수강 학생 뷰를 작성하시오.

create view non_enroll_student as
select s.학번, s.이름, s.학년
from 학생 s
where not exists(
select 1
from 수강 e
where e.학번 = s.학번);

(8)담당 교수별 강의 및 수강 현황 뷰를 작성하시오. 
(과목별 담당 과목 수와 총 수강 학생수를 집계하는 뷰)

create view v_pf_class_current_enroll as
select c.담당교수, count(distinct c.과목코드) as "담당 과목 수", count(e.학번) as "총 수강 학생수"
from 과목 c
left join 수강 e on c.과목코드 = e.과목코드
group by c.담당교수;

(9)학년별 수강 과목 수 및 평균 성적 뷰를 작성하시오. 
(학년에 따른 수강 부담과 성적 추이를 파악하는 뷰 작성)

create view v_year_enroll_avg_score as
select s.학년, count(e.학번) as "수강 과목 수", avg(e.성적) as "평균 성적"
from 학생 s
join 수강 e on e.학번 = s.학번
group by s.학년;

(10)성적 미입력(NULL) 수강 내역 뷰를 작성하시오.
(성적이 아직 입력되지 않은 수강 내역 관리용이다.)

create view v_null_grade as
select *
from 수강
where 성적 is null;

1. 전체 상영 정보 통합 뷰를 작성하시오.
(극장 + 상영관 + 영화 정보를 한눈에 확인 가능하도록 작성)

create view v_all_screen_info as
select t.극장번호, t.극장이름, t.위치, s.상영관번호, s.영화제목, s.가격
from 극장 t
join 상영관 s on t.극장번호 = s.극장번호;

2 .예약 전체 현황 뷰를 작성하시오
(고객이 어떤 극장·상영관·영화를 예약했는지 통합 조회하도록 작성)

create view all_current_reserv as
select c.이름, t.극장이름, s.상영관번호, s.영화제목
from 고객 c
join 예약 r on c.고객번호 = r.고객번호
join 상영관 s on s.극장번호 = r.극장번호
and s.상영관번호 = r.상영관번호
join 극장 t on t.극장번호 = s.극장번호;

3. 극장별 상영 영화 목록 뷰를 작성하시오
(각 극장에서 상영 중인 영화와 가격, 좌석수 조회)

create view v_theater_movie as
select t.극장이름, s.영화제목, s.가격, s.좌석수
from 극장 t
join 상영관 s on t.극장번호 = s.극장번호;

4. 고객별 예약 내역 뷰를 작성하시오
(고객별 예약한 영화 목록과 날짜, 금액 조회가능한 뷰)

create view v_cust_reserv as
select c.이름, s.영화제목, r.날짜, s.가격
from 고객 c
join 예약 r on c.고객번호 = r.고객번호
join 상영관 s on r.극장번호 = s.극장번호
and r.상영관번호 = s.상영관번호

5. 상영관별 예약 인원 및 잔여 좌석 뷰를 작성하시오
(상영관의 총 좌석 대비 예약 현황과 잔여 좌석 계산)

create view v_screen_reserv_seat as
select s.극장번호, s.상영관번호, count(r.고객번호) as "예약 인원", count(r.고객번호) / s.좌석수 as "총 좌석 대비 예약 현황", s.좌석수 - count(r.고객번호) as "잔여 좌석 계산"
from 상영관 s
left join 예약 r on s.극장번호 = r.극장번호
and s.상영관번호 = r.상영관번호
group by s.상영관번호, s.극장번호;

6. 극장별 총 예약 수 및 매출 뷰를 작성하시오
(극장별 총 예약 건수와 누적 매출 집계를 볼 수 있는 뷰)

create view v_theater_reserv_amount as
select t.극장이름, count(r.고객번호) as "예약 건수", sum(s.가격) as 매출
from 극장 t
join 예약 r on t.극장번호 = r.극장번호
join 상영관 s on s.극장번호 = r.극장번호 
and s.상영관번호 = r.상영관번호
group by t.극장번호, t.극장이름;

7. 영화별 예약 현황 뷰를 작성하시오
(영화 제목별 총 예약 인원과 총 매출 집계 뷰)

create view v_current_movie_reserv as
select s.영화제목, count(r.고객번호) as 예약인원, sum(s.가격) as 매출
from 상영관 s
left join 예약 r on r.극장번호 = s.극장번호
and r.상영관번호 = s.상영관번호
group by s.영화제목;

8. 날짜별 예약 현황 뷰를 작성하시오
(날짜별 예약 건수와 해당 날짜 매출 조회 가능한 뷰 작성)

create view v_current_date_reserv as
select r.날짜, count(r.고객번호) as 예약건수, sum(s.가격), 0 as 매출
from 예약 r
join 상영관 s on s.극장번호 = r.극장번호
and s.상영관번호 = r.상영관번호
group by r.날짜;

9. 예약 이력이 없는 고객 뷰
(한 번도 예약한 적 없는 고객 조회하는 뷰 작서)

create view v_non_reserv_cust as
select c.고객번호, c.이름
from 고객 c
where not exists(
select 1
from 예약 r
where r.고객번호 = c.고객번호);

10. 고객별 총 예약 횟수 및 결제 금액 뷰
(고객의 예약 활동 통계) (충성 고객 분석용)

create view v_loyal_cust as
select c.고객번호, c.이름, count(r.고객번호) as 예약횟수, nvl(sum(s.가격), 0) as 결제금액
from 고객 c
left join 예약 r on c.고객번호 = r.고객번호
left join 상영관 s on s.상영관번호 = r.상영관번호
and s.극장번호 = r.극장번호
group by c.고객번호, c.이름;

11. 가격이 가장 비싼 상영관 뷰를 작성하시오
(티켓 가격 기준 상위 상영관 목록 조회하는 뷰 작성)

create view v_high_price_screen as
select 극장번호, 상영관번호, 가격
from 상영관
where 가격 = (
select max(가격)
from 상영관);

12. 위치별 극장 및 상영 현황 뷰
(지역(위치)별로 운영 중인 극장과 상영 영화 수 집계하는 뷰 작성)

create view v_loca_theater_screen as
select t.극장번호, t.극장이름, t.위치, count(s.영화제목) as 영화수
from 극장 t
left join 상영관 s on s.극장번호 = t.극장번호
group by t.극장번호, t.극장이름, t.위치;

13. 만석 상영관 뷰를 작성하시오
(예약 좌석 수가 전체 좌석 수와 동일한 매진 상영관 조회)

create view v_full_seat_screen as
select s.상영관번호, s.극장번호, s.영화제목
from 상영관 s
join 예약 r on s.극장번호 = r.극장번호
and s.상영관번호 = r.상영관번호
group by s.극장번호, s.상영관번호, s.좌석수, s.영화제목
having s.좌석수 - count(r.좌석번호) = 0;

14. 특정 날짜 예약 고객 상세 뷰를 작성하시오
(날짜별 예약 고객 정보와 관람 영화 상세 조회)

create view v_date_reserv_cust as
select r.날짜, c.고객번호, c.이름, c.주소, r.좌석번호, s.극장번호, s.상영관번호, s.영화제목, s.가격
from 예약 r
join 고객 c on c.고객번호 = r.고객번호
join 상영관 s on s.극장번호 = r.극장번호 
and s.상영관번호 = r.상영관번호;

15. 예약 없는 상영관 뷰
개설되었지만 예약이 한 건도 없는 상영관 조회

create view v_non_reserv_screen as

from 상영관 s

select 1

where r.극장번호 = s.극장번호
and r.상영관번호 = s.상영관번호);

1. 전체 주문 통합 현황 뷰를 작성하시오
(주문 + 고객 + 영업사원 정보를 한눈에 통합 조회 뷰)

create or replace view v_all_current_order as
select s.name, s.age, s.salary, o.number, o.custname, o.salesperson, o.amount, c.name, c.city, c.industrytype
from salesperson s
join order o on s.name = o.salesperson
join customer c on c.name = o.custname;

2. 영업사원별 총 주문 금액 및 건수 뷰를 작성하시오
(영업사원별 실적(주문 건수, 총 매출) 집계)

create or replace view v_all_amount_count as
select s.name as 이름, count(o.number) as 주문건수, nvl(sum(o.amount), 0) as 매출
from salesperson s
left join order o on s.name = o.salesperson
group by s.name;

3. 고객별 총 주문 금액 뷰를 작성하시오
(고객별 누적 주문 횟수와 총 구매 금액 집계 포함된 뷰)

create or replace view v_all_order_price as
select c.name as 이름, count(o.custname) as 주문횟수, nvl(sum(o.amount), 0) as 총구매금액
from customer c
left join order o on c.name = o.custname
group by c.name;

4. 도시별 주문 현황 뷰를 작성하시오
(도시(city)별 총 주문 건수와 매출 집계 포함된 뷰)

create or replace view v_city_order as
select c.city as 도시, count(o.number) as 주문건수, nvl(sum(o.amount), 0) as 매출
from customer c
left join order o on c.name = o.custname
group by c.city;

5. 업종별 주문 통계 뷰를 작성하시오
(산업 유형(industrytype)별 주문 건수 및 총 매출 비교 뷰)

create or replace view v_type_order as
select c.industrytype as 산업유형, count(o.number) as 주문건수, nvl(sum(o.amount), 0) as 매출
from customer c
left join order o on c.name = o.custname
group by c.industrytype;

6. 고액 주문 뷰 (amount 상위)를 작성하시오
(주문 금액이 평균 이상인 고액 주문만 조회하는 뷰)

create or replace view v_high_amount as
select o.number as 주문번호,c.name as 고객명, o.amount as 금액
from customer c
join "order" o on c.name = o.custname
where o.amount > (
select avg(amount) 
from "order");

7. 주문 실적 없는 영업사원 뷰를 작성하시오
(한 건도 주문을 처리하지 않은 영업사원 조회 뷰)

create or replace view v_non_salesperson as
select s.name as 이름
from salesperson s
where not exists (
select 1
from order o 
where o.salesperson = s.name);

8. 주문 이력 없는 고객 뷰를 작성하시오
(등록은 되어 있지만 한 번도 주문하지 않은 고객 조회 뷰)

create or replace view v_non_customer as
select c.name as 이름
from customer c
where not exists (
select 1
from order o
where o.custname = c.name);

9. 영업사원 급여 등급 뷰를 작성하시오
(급여 구간별로 영업사원을 분류하는 뷰)

create or replace view v_sales_salary_grade as
select name as 이름, salary as 급여,
case
when salary >= 700 then 'A'
when salary >= 600 then 'B'
when salary >= 500 then 'C'
when salary >= 400 then 'D'
else 'F'
end as 급여등급
from salesperson;

10. 영업사원별 담당 고객 목록 뷰를 작성하시오
(각 영업사원이 주문을 처리한 고객 목록(중복 제거) 생성 뷰)

create or replace view v_sales_cust_list as
select salesperson as 영업사원, custname as 담당고객
from order
group by salesperson, custname;

11. 최고 실적 영업사원 뷰를 작성하시오
(총 주문 금액이 가장 높은 영업사원 조회하는 뷰)

create or replace view v_top_amount_sales as
select salesperson as 이름, sum(amount) as 총실적
from "order"
group by salesperson
having sum(amount) = (
select max(sum(amount))
from "order"
group by salesperson);

12. 도시별 영업사원 활동 현황 뷰를 작성하시오
(어느 도시의 고객에게 얼마나 판매했는지 영업사원 기준 집계 뷰)

create or replace view v_salesperson_current as
select o.salesperson as 사원이름, c.city as 도시, count(o.number) as 판매횟수
from order
join customer c on c.name = o.custname
group by o.salesperson, c.city;

13. 영업사원 급여 대비 매출 효율 뷰를 작성하시오
